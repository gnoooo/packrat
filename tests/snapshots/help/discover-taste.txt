usage: ytm-offline discover taste [-h] [--headers HEADERS] [--output OUTPUT]
                                  [--archive ARCHIVE] [--max-size MAX_SIZE]
                                  [--min-duration MIN_DURATION]
                                  [--max-duration MAX_DURATION]
                                  [--quota QUOTA] [--deep DEEP]
                                  [--adjacent ADJACENT] [--radio RADIO]
                                  [--external EXTERNAL] [--tag TAG]
                                  [--per-album N] [--per-artist PER_ARTIST]
                                  [--no-phase3]

options:
  -h, --help            show this help message and exit
  --quota QUOTA         max tracks per bucket (overrides the per-bucket
                        defaults)
  --deep DEEP           max deep-cut tracks (default: 12)
  --adjacent ADJACENT   max adjacent-artist tracks (default: 10)
  --radio RADIO         max radio tracks (default: 10)
  --external EXTERNAL   max external-graph tracks (default: 5)
  --tag TAG             max tag-search tracks (default: 5)
  --per-album N         max tracks per album/container (default: 2)
  --per-artist PER_ARTIST
                        max tracks per artist (default: 3)
  --no-phase3           do not dedupe against the phase 3 discover manifest

Authentication:
  --headers HEADERS     path to ytmusicapi browser-headers JSON file (env:
                        YTM_HEADERS; default: ~/.config/ytmusic-
                        composer/browser.json)

Run control:
  --output, -o OUTPUT   output root directory (env: YTM_OUTPUT; default:
                        music)
  --archive ARCHIVE     download archive file (env: YTM_ARCHIVE; default:
                        .ytm_offline_archive)
  --max-size MAX_SIZE   skip tracks estimated larger than SIZE (e.g. 3GB)

Selection:
  --min-duration MIN_DURATION
                        minimum track duration in seconds (default: 90)
  --max-duration MAX_DURATION
                        maximum track duration in seconds (default: 540)
